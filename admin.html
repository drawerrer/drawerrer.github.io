${comment.name} on ${docTitle} • ${new Date(comment.date).toLocaleDateString()}

${comment.content}

`; listContainer.appendChild(el); }); }); if (!hasComments) { listContainer.innerHTML = '
No comments found.
'; } } function deleteComment(docTitle, commentId) { if (!confirm('Are you sure you want to delete this comment?')) return; const commentsData = JSON.parse(localStorage.getItem('commentsData')) || {}; if (commentsData[docTitle]) { commentsData[docTitle] = commentsData[docTitle].filter(c => c.id !== commentId); localStorage.setItem('commentsData', JSON.stringify(commentsData)); loadAdminComments(); // Reload UI loadStats(); // Update stats } } // --- WRITINGS LOGIC --- const editModal = document.getElementById('editModal'); let editingId = null; // Title is the ID function loadWritings() { const listBody = document.getElementById('writingsList'); const essays = JSON.parse(localStorage.getItem('essaysData')) || {}; listBody.innerHTML = ''; if (Object.keys(essays).length === 0) { listBody.innerHTML = ` No essays found. `; return; } Object.values(essays).forEach(essay => { const tr = document.createElement('tr'); tr.className = "bg-white dark:bg-zinc-800 border-b border-stone-100 hover:bg-stone-50 transition-colors"; tr.innerHTML = ` ${essay.title} ${essay.date} Published `; listBody.appendChild(tr); }); } function openEditModal(title = null) { editModal.classList.remove('hidden'); editingId = title; if (title) { // Load Existing const essays = JSON.parse(localStorage.getItem('essaysData')); const essay = essays[title]; document.getElementById('editTitle').value = essay.title; document.getElementById('editDate').value = essay.date; document.getElementById('editContent').value = essay.content; // Lock Title for now to keep ID simple document.getElementById('editTitle').readOnly = true; document.getElementById('editTitle').classList.add('bg-stone-100'); } else { // New document.getElementById('editTitle').value = ''; document.getElementById('editDate').value = new Date().toLocaleDateString('ko-KR'); document.getElementById('editContent').value = ''; document.getElementById('editTitle').readOnly = false; document.getElementById('editTitle').classList.remove('bg-stone-100'); } } function closeModal() { editModal.classList.add('hidden'); } function saveEssay() { const title = document.getElementById('editTitle').value.trim(); const date = document.getElementById('editDate').value.trim(); const content = document.getElementById('editContent').value; // Keep formatting if (!title) return alert('Title is required'); const essays = JSON.parse(localStorage.getItem('essaysData')) || {};
